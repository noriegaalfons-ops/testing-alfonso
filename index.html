const ring = document.getElementById('ring'); const numEl = document.getElementById('num'); const cdLabel = document.getElementById('cd-label'); const statusEl = document.getElementById('status'); const statusText = document.getElementById('status-text'); const micIcon = document.getElementById('mic-icon'); const logEl = document.getElementById('log'); const fileInput = document.getElementById('audio-file'); const fileLabel = document.getElementById('file-label'); // ── Helpers ─────────────────────────────────────────────────────────────── function log(msg, cls = '') { const line = document.createElement('span'); line.className = 'log-line' + (cls ? ' ' + cls : ''); const t = new Date().toLocaleTimeString('es', { hour12: false }); line.textContent = `[${t}] ${msg}`; logEl.appendChild(line); logEl.scrollTop = logEl.scrollHeight; } function setStatus(txt, cls) { statusEl.className = 'status ' + (cls || ''); statusText.textContent = txt; } function actualizarRing(s, tot, color) { const offset = tot > 0 ? CIRCUM * (s / tot) : CIRCUM; ring.style.strokeDashoffset = offset; ring.style.stroke = color || 'var(--accent)'; numEl.textContent = s > 0 ? s : (s === 0 ? '0' : '—'); } // ── Audio Context ───────────────────────────────────────────────────────── function getCtx() { if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)(); if (audioCtx.state === 'suspended') audioCtx.resume(); return audioCtx; } // Al elegir archivo: detener reconocimiento, cargar, reiniciar reconocimiento fileInput.addEventListener('change', async (e) => { const file = e.target.files[0]; if (!file) return; // Pausar reconocimiento para liberar micrófono durante la carga detenerReconocimiento(); log('Cargando archivo...', 'hi'); try { const ctx = getCtx(); audioBuffer = await ctx.decodeAudioData(await file.arrayBuffer()); fileLabel.textContent = '🎵 ' + file.name; fileLabel.style.color = 'var(--green)'; log('Audio listo: ' + file.name, 'ok'); } catch (err) { log('Error al cargar: ' + err.message, 'err'); audioBuffer = null; } // Reiniciar micrófono después de cargar setTimeout(() => iniciarReconocimiento(), 400); }); // ── Reproducción ────────────────────────────────────────────────────────── function reproducir(offset = 0) { const ctx = getCtx(); gainNode = ctx.createGain(); gainNode.gain.setValueAtTime(0, ctx.currentTime); gainNode.gain.linearRampToValueAtTime(1, ctx.currentTime + 0.3); gainNode.connect(ctx.destination); if (audioBuffer) { sourceNode = ctx.createBufferSource(); sourceNode.buffer = audioBuffer; sourceNode.loop = true; sourceNode.connect(gainNode); sourceNode.start(0, offset % audioBuffer.duration); } else { sourceNode = ctx.createOscillator(); const lfo = ctx.createOscillator(); const lfoG = ctx.createGain(); lfo.frequency.setValueAtTime(4, ctx.currentTime); lfoG.gain.setValueAtTime(8, ctx.currentTime); lfo.connect(lfoG); lfoG.connect(sourceNode.frequency); sourceNode.type = 'sine'; sourceNode.frequency.setValueAtTime(528, ctx.currentTime); sourceNode.connect(gainNode); lfo.start(); sourceNode.start(); } playStart = ctx.currentTime - offset; } function pausarAudio() { if (!sourceNode) return; const ctx = getCtx(); audioOffset = ctx.currentTime - playStart; gainNode.gain.setValueAtTime(gainNode.gain.value, ctx.currentTime); gainNode.gain.linearRampToValueAtTime(0, ctx.currentTime + 0.25); setTimeout(() => { try { sourceNode.stop(); } catch(_) {} sourceNode = null; }, 300); } // ── Flujo principal ─────────────────────────────────────────────────────── function iniciarEspera() { fase = 'wait'; seg = WAIT_TOTAL; total = WAIT_TOTAL; segRestantes = ON_TOTAL; // reinicio completo audioOffset = 0; actualizarRing(seg, total, 'var(--yellow)'); cdLabel.textContent = 'para sonar'; setStatus('Preparando...', 'waiting'); micIcon.className = 'mic-icon detected'; document.body.classList.remove('playing'); log('Palabra detectada — reproduciendo en ' + WAIT_TOTAL + ' s', 'hi'); timer = setInterval(() => { seg--; actualizarRing(seg, total, seg <= 2 ? '#ff9933' : 'var(--yellow)'); if (seg <= 0) { clearInterval(timer); timer = null; iniciarReproduccion(0); } }, 1000); } function iniciarReproduccion(offset) { fase = 'playing'; seg = segRestantes; total = ON_TOTAL; actualizarRing(seg, total, 'var(--accent)'); cdLabel.textContent = 'sonando'; setStatus('Reproduciendo ♪', 'playing'); document.body.classList.add('playing'); micIcon.className = 'mic-icon listening'; log('Audio reproduciendo — ' + seg + ' s restantes', 'ok'); reproducir(offset); // Mantener reconocimiento activo para escuchar "pausa" iniciarReconocimiento(); timer = setInterval(() => { seg--; segRestantes = seg; actualizarRing(seg, total, seg <= 5 ? 'var(--accent2)' : 'var(--accent)'); if (seg <= 0) { clearInterval(timer); timer = null; finalizarReproduccion(); } }, 1000); } function pausarPorVoz() { if (fase !== 'playing') return; clearInterval(timer); timer = null; pausarAudio(); fase = 'paused'; document.body.classList.remove('playing'); cdLabel.textContent = 'pausado'; numEl.textContent = segRestantes; ring.style.stroke = 'var(--muted)'; setStatus('Pausado ■ · Di "play"', 'paused'); micIcon.className = 'mic-icon listening'; log('Pausado. Quedan ' + segRestantes + ' s. Di "play" para continuar.', 'warn'); } function reanudarPorVoz() { if (fase !== 'paused' || segRestantes <= 0) return; log('Reanudando desde ' + segRestantes + ' s restantes...', 'hi'); iniciarReproduccion(audioOffset); } function finalizarReproduccion() { pausarAudio(); document.body.classList.remove('playing'); fase = 'idle'; segRestantes = ON_TOTAL; audioOffset = 0; numEl.textContent = '✓'; cdLabel.textContent = 'finalizado'; ring.style.strokeDashoffset = CIRCUM; ring.style.stroke = 'var(--muted)'; setStatus('Listo ✓ · Di "Saturno"', 'paused'); micIcon.className = 'mic-icon listening'; log('Audio finalizado. Di "Saturno" para repetir.', 'ok'); iniciarReconocimiento(); } // ── Reconocimiento de voz ───────────────────────────────────────────────── function detenerReconocimiento() { if (recognition) { try { recognition.abort(); } catch(_) {} recognition = null; } } function iniciarReconocimiento() { const SR = window.SpeechRecognition || window.webkitSpeechRecognition; if (!SR) { log('SpeechRecognition no disponible en este navegador', 'err'); setStatus('Sin soporte de voz', 'error'); micIcon.className = 'mic-icon error'; return; } // Evitar instancias duplicadas detenerReconocimiento(); recognition = new SR(); recognition.lang = 'es-ES'; recognition.continuous = false; recognition.interimResults = true; recognition.onstart = () => { if (fase !== 'playing') { setStatus('Escuchando...', 'listening'); micIcon.className = 'mic-icon listening'; } }; recognition.onresult = (e) => { for (let i = e.resultIndex; i < e.results.length; i++) { const texto = e.results[i][0].transcript.toLowerCase().trim(); // Comandos if (texto.includes('saturno') && (fase === 'idle' || fase === 'paused' && segRestantes === ON_TOTAL)) { log('Comando: "saturno"', 'hi'); detenerReconocimiento(); getCtx(); iniciarEspera(); return; } if (texto.includes('pausa') && fase === 'playing') { log('Comando: "pausa"', 'hi'); detenerReconocimiento(); pausarPorVoz(); setTimeout(() => iniciarReconocimiento(), 400); return; } if ((texto.includes('play') || texto.includes('continuar') || texto.includes('reanudar')) && fase === 'paused') { log('Comando: "play"', 'hi'); detenerReconocimiento(); reanudarPorVoz(); return; } } }; recognition.onerror = (e) => { if (e.error === 'no-speech' || e.error === 'aborted') { setTimeout(() => { if (fase === 'idle' || fase === 'paused' || fase === 'playing') iniciarReconocimiento(); }, 300); return; } log('Error de voz: ' + e.error, 'err'); if (e.error === 'not-allowed') { setStatus('Micrófono bloqueado', 'error'); micIcon.className = 'mic-icon error'; log('→ Permite el micrófono en Chrome y recarga', 'err'); } }; recognition.onend = () => { setTimeout(() => { if (fase === 'idle' || fase === 'paused' || fase === 'playing') iniciarReconocimiento(); }, 200); }; try { recognition.start(); } catch(_) { setTimeout(() => iniciarReconocimiento(), 500); } } window.addEventListener('load', () => { iniciarReconocimiento(); }); Sonido · Saturno
